#include <QtWidgets>
#include <QtConcurrent/qtconcurrentrun.h>
#include <QtDataVisualization/Q3DSurface>
#include <QtDataVisualization/QSurface3DSeries>
#include <QtDataVisualization/QSurfaceDataProxy>
#include <cmath>
#include <vector>
using namespace QtDataVisualization;

// ---------- Расстояние от точки до отрезка ----------
double pointToSegmentDistance(const QPointF &p, const QPointF &a, const QPointF &b) {
    double dx = b.x() - a.x();
    double dy = b.y() - a.y();
    if (dx == 0 && dy == 0)
        return std::hypot(p.x() - a.x(), p.y() - a.y());

    double t = ((p.x() - a.x()) * dx + (p.y() - a.y()) * dy) / (dx * dx + dy * dy);
    t = std::max(0.0, std::min(1.0, t));
    double projx = a.x() + t * dx;
    double projy = a.y() + t * dy;
    return std::hypot(p.x() - projx, p.y() - projy);
}

// ---------- Построение матрицы ----------
std::vector<std::vector<int> > buildMatrix(const std::vector<QPointF> &polyline,
                                           double R, int gridW, int gridH,
                                           QRectF &bboxOut) {
    QRectF bbox(polyline[0], QSizeF(1, 1));
    for (auto &pt: polyline)
        bbox = bbox.united(QRectF(pt.x() - R, pt.y() - R, 2 * R, 2 * R));
    bboxOut = bbox;

    std::vector<std::vector<int> > matrix(gridH, std::vector<int>(gridW, 0));
    double cellW = bbox.width() / gridW;
    double cellH = bbox.height() / gridH;

    for (int y = 0; y < gridH; ++y) {
        for (int x = 0; x < gridW; ++x) {
            QPointF pt(bbox.left() + (x + 0.5) * cellW,
                       bbox.top() + (y + 0.5) * cellH);
            int count = 0;
            for (size_t i = 0; i + 1 < polyline.size(); ++i) {
                double dist = pointToSegmentDistance(pt, polyline[i], polyline[i + 1]);
                if (dist <= R) ++count;
            }
            matrix[y][x] = count;
        }
    }
    return matrix;
}

// ---------- 2D Heatmap ----------
class HeatmapView : public QWidget {
public:
    HeatmapView(QWidget *parent = nullptr) : QWidget(parent) {
    }

    void setMatrix(const std::vector<std::vector<int> > &m, const QRectF &bbox) {
        matrix_ = m;
        bbox_ = bbox;
        update();
    }

protected:
    void paintEvent(QPaintEvent *) override {
        if (matrix_.empty()) return;
        QPainter p(this);
        int gridH = matrix_.size();
        int gridW = matrix_[0].size();
        double cellW = bbox_.width() / gridW;
        double cellH = bbox_.height() / gridH;

        int maxVal = 0;
        for (auto &row: matrix_)
            for (int v: row)
                maxVal = std::max(maxVal, v);

        for (int y = 0; y < gridH; ++y) {
            for (int x = 0; x < gridW; ++x) {
                int v = matrix_[y][x];
                if (v <= 0) continue;
                double t = maxVal > 0 ? double(v) / maxVal : 0.0;
                QColor c = QColor::fromHsvF(0.66 - 0.66 * t, 1.0, 1.0);
                p.fillRect(QRectF(bbox_.left() + x * cellW,
                                  bbox_.top() + y * cellH,
                                  cellW, cellH), c);
            }
        }
    }

private:
    std::vector<std::vector<int> > matrix_;
    QRectF bbox_;
};

// ---------- 3D Surface ----------
class Surface3DView : public QWidget {
public:
    Surface3DView(QWidget *parent = nullptr) : QWidget(parent) {
        surface_ = new Q3DSurface();
        container_ = QWidget::createWindowContainer(surface_);
        auto layout = new QVBoxLayout(this);
        layout->setContentsMargins(0, 0, 0, 0);
        layout->addWidget(container_);
    }

    void setMatrix(const std::vector<std::vector<int> > &matrix) {
        QSurfaceDataArray *dataArray = new QSurfaceDataArray;
        dataArray->reserve(matrix.size());
        for (int y = 0; y < (int) matrix.size(); ++y) {
            QSurfaceDataRow *row = new QSurfaceDataRow(matrix[y].size());
            for (int x = 0; x < (int) matrix[y].size(); ++x)
                (*row)[x].setPosition(QVector3D(x, matrix[y][x], y));
            *dataArray << row;
        }
        QSurfaceDataProxy *proxy = new QSurfaceDataProxy();
        proxy->resetArray(dataArray);
        QSurface3DSeries *series = new QSurface3DSeries(proxy);
        series->setDrawMode(QSurface3DSeries::DrawSurface);
        series->setFlatShadingEnabled(false);
        surface_->addSeries(series);
        surface_->activeTheme()->setColorStyle(Q3DTheme::ColorStyleRangeGradient);
        surface_->activeTheme()->setBaseColors({Qt::blue, Qt::cyan, Qt::green, Qt::yellow, Qt::red});
    }

private:
    Q3DSurface *surface_;
    QWidget *container_;
};

// ---------- Основной виджет ----------
class MatrixWidget : public QWidget {
    Q_OBJECT

public:
    MatrixWidget(QWidget *parent = nullptr)
        : QWidget(parent), R_(25.0), gridW_(80), gridH_(40) {
        polyline_ = {{50, 150}, {200, 100}, {350, 180}, {450, 80}};

        auto mainLayout = new QVBoxLayout(this);

        // --- Панель управления ---
        auto controls = new QHBoxLayout;
        radiusSpin_ = new QSpinBox;
        radiusSpin_->setRange(1, 100);
        radiusSpin_->setValue(static_cast<int>(R_));
        controls->addWidget(new QLabel("Radius:"));
        controls->addWidget(radiusSpin_);

        gridWSpin_ = new QSpinBox;
        gridWSpin_->setRange(10, 300);
        gridWSpin_->setValue(gridW_);
        controls->addWidget(new QLabel("Grid W:"));
        controls->addWidget(gridWSpin_);

        gridHSpin_ = new QSpinBox;
        gridHSpin_->setRange(10, 300);
        gridHSpin_->setValue(gridH_);
        controls->addWidget(new QLabel("Grid H:"));
        controls->addWidget(gridHSpin_);

        auto recalcBtn = new QPushButton("Пересчитать");
        controls->addWidget(recalcBtn);

        toggleBtn_ = new QPushButton("2D / 3D");
        controls->addWidget(toggleBtn_);
        mainLayout->addLayout(controls);

        // --- Надпись загрузки ---
        loadingLabel_ = new QLabel("Loading…");
        loadingLabel_->setAlignment(Qt::AlignCenter);
        mainLayout->addWidget(loadingLabel_);

        // --- Виджеты отображения ---
        stacked_ = new QStackedLayout;
        heatmap_ = new HeatmapView;
        surface3D_ = new Surface3DView;
        stacked_->addWidget(heatmap_);
        stacked_->addWidget(surface3D_);
        mainLayout->addLayout(stacked_);

        futureWatcher_ = new QFutureWatcher<std::vector<std::vector<int> > >(this);
        connect(futureWatcher_, &QFutureWatcherBase::finished, this, &MatrixWidget::onMatrixReady);
        connect(recalcBtn, &QPushButton::clicked, this, &MatrixWidget::startAsyncCalculation);
        connect(toggleBtn_, &QPushButton::clicked, this, &MatrixWidget::toggleView);

        startAsyncCalculation();
    }

private slots:
    void onMatrixReady() {
        matrix_ = futureWatcher_->result();
        loadingLabel_->hide();
        heatmap_->setMatrix(matrix_, bbox_);
        surface3D_->setMatrix(matrix_);
    }

    void startAsyncCalculation() {
        R_ = radiusSpin_->value();
        gridW_ = gridWSpin_->value();
        gridH_ = gridHSpin_->value();
        loadingLabel_->show();
        matrix_.clear();
        QFuture<std::vector<std::vector<int> > > future = QtConcurrent::run([=]() {
            return buildMatrix(polyline_, R_, gridW_, gridH_, bbox_);
        });
        futureWatcher_->setFuture(future);
    }

    void toggleView() {
        int current = stacked_->currentIndex();
        stacked_->setCurrentIndex(1 - current); // переключение между 0 и 1
    }

private:
    std::vector<QPointF> polyline_;
    std::vector<std::vector<int> > matrix_;
    QRectF bbox_;
    double R_;
    int gridW_, gridH_;

    QSpinBox *radiusSpin_, *gridWSpin_, *gridHSpin_;
    QPushButton *toggleBtn_;
    QLabel *loadingLabel_;
    QFutureWatcher<std::vector<std::vector<int> > > *futureWatcher_;

    QStackedLayout *stacked_;
    HeatmapView *heatmap_;
    Surface3DView *surface3D_;
};

int main(int argc, char **argv) {
    QApplication app(argc, argv);
    MatrixWidget w;
    w.resize(1000, 700);
    w.setWindowTitle("Матрица по расстоянию до ломаной — 2D/3D");
    w.show();
    return app.exec();
}
